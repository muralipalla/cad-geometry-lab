from bokeh.plotting import figure
from bokeh.models import PointDrawTool

plot = figure(
    title="Draggable control points",
    match_aspect=True
)

renderer = plot.scatter(
    x="x", y="y",
    source=source, size=14
)

plot.add_tools(PointDrawTool(
    renderers=[renderer]
))
